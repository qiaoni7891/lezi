local AAC = (function()
	local S = {
		enabled = true, antiKick = true, antiRemoteReport = true, antiSpeedDetect = true,
		antiTeleportDetect = true, hideHooks = true, kickCount = 0, blockedReports = 0,
		_speedCap = 500, _smoothThreshold = 40,
	}
	local function installNamecallGuard()
		if not getrawmetatable or not newcclosure then return end
		local mt = getrawmetatable(game)
		if not mt or not mt.__namecall then return end
		local oldNamecall = mt.__namecall
		setreadonly(mt, false)
		mt.__namecall = newcclosure(function(obj, ...)
			local method = getnamecallmethod()
			local args = {...}
			if S.antiKick and method == "Kick" then
				S.kickCount = S.kickCount + 1
				warn("[AAC] 已拦截踢出请求 #" .. tostring(S.kickCount))
				return nil
			end
			if S.antiRemoteReport and (method == "FireServer" or method == "InvokeServer") then
				local tag = args[1]
				if type(tag) == "string" then
					local lc = string.lower(tag)
					if lc:find("anticheat") or lc:find("ac_") or lc:find("detect")
						or lc:find("report") or lc:find("flag")
						or lc:find("suspect") or lc:find("cheat")
						or lc:find("crash") or lc:find("ban")
						or lc:find("fly") or lc:find("teleport")
						or lc:find("speed") or lc:find("movement")
						or lc:find("exploit") or lc:find("hack")
						or lc == "kick" or lc == "ac" then
						S.blockedReports = S.blockedReports + 1
						return nil
					end
				end
			end
			return oldNamecall(obj, ...)
		end)
		setreadonly(mt, true)
	end
	pcall(installNamecallGuard)
	if RunService then
		pcall(function()
			RunService.Heartbeat:Connect(function()
				if not S.antiSpeedDetect then return end
				local char = LocalPlayer and LocalPlayer.Character
				if not char then return end
				local hum = char:FindFirstChildOfClass("Humanoid")
				if not hum then return end
				if hum.WalkSpeed > S._speedCap then hum.WalkSpeed = S._speedCap end
				if hum.JumpPower > S._speedCap then hum.JumpPower = S._speedCap end
			end)
		end)
	end
	local pendingTeleport = nil
	if RunService then
		pcall(function()
			RunService.PreSimulation:Connect(function()
				if not S.antiTeleportDetect then return end
				local char = LocalPlayer and LocalPlayer.Character
				if not char then pendingTeleport = nil; return end
				local root = char:FindFirstChild("HumanoidRootPart")
				if not root then return end
				if pendingTeleport then
					local dist = (root.Position - pendingTeleport).Magnitude
					if dist < 5 then
						pendingTeleport = nil
					else
						local dir  = (pendingTeleport - root.Position).Unit
						local step = math.min(S._smoothThreshold, dist)
						root.CFrame = root.CFrame + dir * step
						root.AssemblyLinearVelocity   = Vector3.zero
						root.AssemblyAngularVelocity  = Vector3.zero
					end
				end
			end)
		end)
	end
	if S.hideHooks and debug then
		pcall(function()
			if debug.getinfo then
				local oldGI = debug.getinfo
				debug.getinfo = newcclosure(function(...)
					local info = oldGI(...)
					if type(info) == "table" and info.source then
						local src = tostring(info.source)
						if src:find("AAC") or src:find("乔尼") or src:find("JoJo") or src:find("KeyAuth") then
							info.source = "=[C]"; info.short_src = "[C]"; info.name = "?"
						end
					end
					return info
				end)
			end
		end)
	end
	pcall(function()
		if type(getgenv) == "function" then
			local env = getgenv()
			if type(env) == "table" then env._AAC_ACTIVE = nil end
		end
	end)
	return S
end)()

local function GetRainbowColor(speed)
	speed = speed or 5

end

	if name == "彩虹色" then return GetRainbowColor() end
	return CommonColors[name] or Color3.fromRGB(255, 0, 0)

local function GetCharacter(player)
	player = player or LocalPlayer
	local character = player.Character
	if not character then return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart") or character:FindFirstChild("Torso") or character:FindFirstChild("UpperTorso")
	if not humanoid or not root then return end
	return character, humanoid, root
end
local function IsAlive(player)
	local _, humanoid = GetCharacter(player)
	return humanoid ~= nil and humanoid.Health > 0
end
local function GetPlayerJob(player)
	return player and player.Team and player.Team.Name or "Civilian"
end
local function EquipWeapon()
	local character, humanoid = GetCharacter(LocalPlayer)
	if not character or not humanoid then return end
	local current = character:FindFirstChildOfClass("Tool")
	if current then return current end
	local backpack = LocalPlayer:FindFirstChild("Backpack")
	if not backpack then return end
	for _, item in ipairs(backpack:GetChildren()) do
		if item:IsA("Tool") then
			humanoid:EquipTool(item)
			task.wait(0.1)
			return character:FindFirstChildOfClass("Tool")
		end
	end
end

local function applyBorderColor(color, rainbow)
	local mainFrame = mainWindow and mainWindow.UIElements and mainWindow.UIElements.Main
	if not mainFrame then return end

	if not stroke then return end

			ColorSequenceKeypoint.new(0, Color3.fromHex("FF0000")), ColorSequenceKeypoint.new(0.16, Color3.fromHex("FFA500")),

			ColorSequenceKeypoint.new(1, Color3.fromHex("EE82EE")),
		})

			if gradient.Parent then gradient.Rotation = (gradient.Rotation + 1.5) % 360 end
		end)

	else
		local c = color or Color3.new(1, 1, 1)
		stroke.Color = c; gradient.Color = ColorSequence.new(c); gradient.Rotation = 0
		settings.isBorderRainbow = false; settings.borderColor = nil

local function safeTeleport(pos)
	local character, _, root = GetCharacter(LocalPlayer)
	if not character or not root or not pos then return end
	local targetCF = CFrame.new(pos + Vector3.new(0, 3, 0))
	if AAC and AAC.antiTeleportDetect then
		local currentPos = root.Position
		local distance = (targetCF.Position - currentPos).Magnitude
		if distance > 100 then
			local steps = math.ceil(distance / 50)
			for i = 1, steps do
				local lerpPos = currentPos:Lerp(targetCF.Position, i / steps)
				root.CFrame = CFrame.new(lerpPos + Vector3.new(0, 3, 0))
				task.wait(0.01)
			end
		else
			character:PivotTo(targetCF)
		end
	else
		character:PivotTo(targetCF)
	end
	pcall(function()
		if PlayerEvent then
			local id = ((character:GetAttribute("CharPivotToId") or 0) + 1) % 100
			character:SetAttribute("CharPivotToId", id)
			PlayerEvent:FireServer("charPivotTo", targetCF, character, id)
		end
	end)
end

local function getFlyDirection()
	local cam = Workspace.CurrentCamera
	if not cam then return Vector3.zero end
	local moveVec = Vector3.zero
	if PlayerControls then
		local mv = PlayerControls:GetMoveVector()
		moveVec = cam.CFrame.LookVector * -mv.Z + cam.CFrame.RightVector * mv.X
	else
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + cam.CFrame.RightVector end
	end
	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVec = moveVec + Vector3.new(0, 1, 0) end
	if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then moveVec = moveVec + Vector3.new(0, -1, 0) end
	if moveVec.Magnitude > 0 then moveVec = moveVec.Unit end
	return moveVec
end

local function flyDetectWall()
	local _, _, root = GetCharacter(LocalPlayer)
	if not root then return false end
	local pos = root.Position
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { LocalPlayer.Character }
	local rayCount = 8
	local rayLength = 3.5
	for i = 1, rayCount do
		local angle = (i / rayCount) * 2 * math.pi
		local dx = math.cos(angle)
		local dz = math.sin(angle)
		for j = -1, 1 do
			local dir = Vector3.new(dx, j * 0.5, dz).Unit
			local result = Workspace:Raycast(pos, dir * rayLength, params)
			if result and result.Instance and result.Instance.CanCollide and result.Instance.Transparency < 0.9 then
				return true
			end
		end
	end
	return false
end

local function startPhysicalFly()
	clearPhysicalFly()
	local _, hum, root = GetCharacter(LocalPlayer)
	if not root or not hum then return end
	PlayerConfig.flyEnabled = true
	secureNetworkOwnership()
	disguiseHumanoidState(true)
	
	local bv = Instance.new("BodyVelocity")
	bv.Name = "圣奥里_FlyVelocity"
	bv.MaxForce = Vector3.new(PlayerConfig.flyMaxForce, PlayerConfig.flyMaxForce, PlayerConfig.flyMaxForce)
	bv.Velocity = Vector3.zero
	bv.Parent = root
	PlayerRuntime.bodyVelocity = bv
	
	local bg = Instance.new("BodyGyro")
	bg.Name = "圣奥里_FlyGyro"
	bg.MaxTorque = Vector3.new(PlayerConfig.flyMaxForce, PlayerConfig.flyMaxForce, PlayerConfig.flyMaxForce)
	bg.P = 5e3
	bg.Parent = root
	PlayerRuntime.bodyGyro = bg
	
	hum.PlatformStand = true
	hum.AutoRotate = false
	
	if PlayerConfig.healthLockEnabled then startHealthLock() end
	PlayerRuntime.diedConn = hum.Died:Connect(function()
		if PlayerConfig.flyEnabled and hum then
			pcall(function()
				hum.Health = hum.MaxHealth
				hum:ChangeState(Enum.HumanoidStateType.Running)
			end)
		end
	end)
	
	local currentSpeed = 0
	PlayerRuntime.flyConn = RunService.RenderStepped:Connect(function(dt)
		if not PlayerConfig.flyEnabled then return end
		local _, currentHum, currentRoot = GetCharacter(LocalPlayer)
		local cam = Workspace.CurrentCamera
		if not currentRoot or not currentHum or not cam then return end
		if PlayerRuntime.bodyVelocity and PlayerRuntime.bodyGyro then
			local speedMul = 1
			if PlayerConfig.wallDetectEnabled then
				PlayerRuntime.wallDetected = flyDetectWall()
				if PlayerRuntime.wallDetected then speedMul = 0.3 end
			end
			
			local targetSpeed = math.min(PlayerConfig.flySpeed, 60) * speedMul
			if currentSpeed < targetSpeed then 
				currentSpeed = currentSpeed + 2 * dt 
			else 
				currentSpeed = targetSpeed 
			end
			
			local direction = getFlyDirection()
			local jitter = Vector3.new(math.random(-2, 2), math.random(-2, 2), math.random(-2, 2))
			local finalVelocity = (direction * currentSpeed) + (direction.Magnitude > 0 and jitter * 0.1 or Vector3.zero)
			
			PlayerRuntime.bodyVelocity.Velocity = finalVelocity
			PlayerRuntime.bodyGyro.CFrame = cam.CFrame
			
			if currentHum:GetState() == Enum.HumanoidStateType.Freefall then 
				currentHum:ChangeState(Enum.HumanoidStateType.Running) 
			end
		end
	end)
	updatePlayerQuick()
end

local function targetAllowed(player, onlyPolice, onlyCivilian)
	if not player or player == LocalPlayer then return false end
	if onlyPolice then return player.Team and player.Team.Name == "Police" end
	if onlyCivilian then return player.Team and player.Team.Name == "Civilian" end

	if not enabled then return true end
	return player:GetAttribute("CombatMode") == true or player:GetAttribute("Pursuit") == true

local auraLast = 0
RunService.Heartbeat:Connect(function()
	if not CombatConfig.auraEnabled or not PlayerEvent then return end
	local now = tick()
	if now - auraLast < CombatConfig.auraInterval then return end
	local _, _, myRoot = GetCharacter(LocalPlayer)
	if not myRoot then return end
	local nearest, nearestDist
	for _, player in ipairs(Players:GetPlayers()) do
		if targetAllowed(player, CombatConfig.auraOnlyPolice, CombatConfig.auraOnlyCivilian) and IsAlive(player) and inCombat(player, CombatConfig.auraCombatCheck) then
			local _, _, root = GetCharacter(player)
			if root then
				local dist = (root.Position - myRoot.Position).Magnitude
				if dist <= CombatConfig.auraRange and (not nearestDist or dist < nearestDist) then
					nearestDist = dist; nearest = player
				end
			end
		end
	end
	if nearest then
		local _, _, root = GetCharacter(nearest)
		local myPos = myRoot.Position
		pcall(function()
			PlayerEvent:FireServer("damage", {
				bodyParts = { { "Head", 1 } },
				shotCode = { myPos, (root.Position - myPos).Unit },
				pos = root.Position, target = nearest,
				damageFactor = CombatConfig.auraDamage, bulletProofTool = false,
			})
		end)
		auraLast = now
	end
end)

local function aimTargetPart(char)
	for _, name in ipairs(PartMap[AimConfig.targetPart] or { "Head" }) do
		local part = char:FindFirstChild(name)
		if part then return part end
	end
	return char:FindFirstChild("HumanoidRootPart")

local function aimVisible(part)
	if not AimConfig.wallCheck then return true end
	local camera = Workspace.CurrentCamera
	local params = RaycastParams.new()
	params.FilterDescendantsInstances = { LocalPlayer.Character, camera }
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.IgnoreWater = true
	local hit = Workspace:Raycast(camera.CFrame.Position, part.Position - camera.CFrame.Position, params)
	return not hit or hit.Instance:IsDescendantOf(part.Parent)
end
local function getBestAimTarget()
	local camera = Workspace.CurrentCamera
	if not camera then return end
	local center = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)
	local best, bestValue
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and IsAlive(player) and targetAllowed(player, AimConfig.onlyPolice, AimConfig.onlyCivilian) and inCombat(player, AimConfig.combatCheck) then
			if AimConfig.teamCheck and LocalPlayer.Team and player.Team == LocalPlayer.Team then continue end
			if AimConfig.friendCheck then
				local ok, friend = pcall(function() return LocalPlayer:IsFriendsWith(player.UserId) end)
				if ok and friend then continue end
			end
			local char = player.Character
			local part = char and aimTargetPart(char)
			local _, hum, root = GetCharacter(player)
			if part and hum and root and aimVisible(part) then
				local screen, onScreen = camera:WorldToViewportPoint(part.Position)
				if onScreen then
					local fov = (Vector2.new(screen.X, screen.Y) - center).Magnitude
					if fov <= AimConfig.fov then
						local score
						if AimConfig.targetMode == "距离最近" then
							local _, _, myRoot = GetCharacter(LocalPlayer)
							score = myRoot and (myRoot.Position - root.Position).Magnitude or math.huge
						elseif AimConfig.targetMode == "血量最低" then score = hum.Health
						else score = fov end
						if not bestValue or score < bestValue then
							bestValue = score
							best = { player = player, part = part, screen = screen }
						end
					end
				end
			end
		end
	end
	return best
end

	if AimConfig.enabled then

			local targetPos = target.part.Position

			local targetCF = CFrame.new(camera.CFrame.Position, targetPos)
			camera.CFrame = AimConfig.smoothness >= 1 and targetCF or camera.CFrame:Lerp(targetCF, AimConfig.smoothness)
		end

local HitboxConfig = {
	active = false, size = 10, transparency = 0.7, teamCheck = false,
	color = "红色", material = "Neon", rainbow = false, checkCorpses = false,
	outline = false, collision = false, glow = false, pulse = false, affectNPC = false,
}
local hitboxOriginal = {}
local function resetHitbox(char)
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local orig = hitboxOriginal[root]
	if orig then
		root.Size = orig.Size; root.Transparency = orig.Transparency
		root.Material = orig.Material; root.CanCollide = orig.CanCollide; root.Color = orig.Color
	end
	local h = root:FindFirstChild("乔尼_HitboxHighlight"); if h then h:Destroy() end
	local l = root:FindFirstChild("乔尼_HitboxLight"); if l then l:Destroy() end
end
local function applyHitbox(char)
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	if not hitboxOriginal[root] then
		hitboxOriginal[root] = { Size = root.Size, Transparency = root.Transparency, Material = root.Material, CanCollide = root.CanCollide, Color = root.Color }
	end
	if not HitboxConfig.active then resetHitbox(char); return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if HitboxConfig.checkCorpses and hum and hum.Health <= 0 then resetHitbox(char); return end
	local size = HitboxConfig.size
	if HitboxConfig.pulse then size = size * (math.sin(tick() * 2) * 0.2 + 1) end
	root.Size = Vector3.new(size, size, size)
	root.Transparency = HitboxConfig.transparency
	root.Material = Enum.Material[HitboxConfig.material] or Enum.Material.Neon
	root.CanCollide = HitboxConfig.collision
	root.Color = HitboxConfig.rainbow and GetRainbowColor(5) or GetColor(HitboxConfig.color)
	if HitboxConfig.outline then
		local hl = root:FindFirstChild("乔尼_HitboxHighlight") or Instance.new("Highlight")
		hl.Name = "乔尼_HitboxHighlight"; hl.FillTransparency = 1; hl.OutlineColor = root.Color
		hl.OutlineTransparency = HitboxConfig.transparency; hl.Parent = root
	else
		local hl = root:FindFirstChild("乔尼_HitboxHighlight"); if hl then hl:Destroy() end
	end
	if HitboxConfig.glow then
		local light = root:FindFirstChild("乔尼_HitboxLight") or Instance.new("PointLight")
		light.Name = "乔尼_HitboxLight"; light.Brightness = 5; light.Range = 15; light.Color = root.Color; light.Parent = root
	else
		local light = root:FindFirstChild("乔尼_HitboxLight"); if light then light:Destroy() end
	end
end
RunService.Heartbeat:Connect(function()
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			if HitboxConfig.teamCheck and LocalPlayer.Team and player.Team == LocalPlayer.Team then resetHitbox(player.Character)
			else applyHitbox(player.Character) end
		end
	end
	if HitboxConfig.affectNPC then
		for _, obj in ipairs(Workspace:GetDescendants()) do
			if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(obj) then applyHitbox(obj) end
		end
	end
end)

local FL = {}

FL.DMG = { Enabled = true, RayMode = "near", LeadStuds = 1.5, SendBullet = true }

function FL.Equipped()
	local char = LocalPlayer.Character
	if not char then return nil end
	local tool = char:FindFirstChildOfClass("Tool")
	if tool then return tool end
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("ModuleScript") and d.Name == "Config" then return d.Parent end
	end
	return nil
end

function FL.WeaponInfo()
	local w = FL.Equipped()
	if not w then return nil end
	local cfg = w:FindFirstChild("Config")
	if not cfg or not cfg:IsA("ModuleScript") then return nil end
	local ok, data = pcall(require, cfg)
	if not ok or type(data) ~= "table" then return nil end
	return { model = w, name = w.Name, damage = tonumber(data.DAMAGE) or 5,
		category = data.CATEGORY, melee = data.CATEGORY == "Melee" }
end

function FL.Hit(target)
	if not FL.DMG.Enabled or not target or not PlayerEvent then return false end
	local char = (typeof(target) == "Instance" and target:IsA("Player")) and target.Character or target
	if not char or not char.Parent then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return false end
	local part = char:FindFirstChild("Head") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("HumanoidRootPart")
	if not part then return false end
	local pos = part.Position
	local eye = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	local origin = eye and eye.Position or pos
	local dir = pos - origin
	if dir.Magnitude < 0.01 then dir = Vector3.new(0,0,-1) else dir = dir.Unit end
	local from = FL.DMG.RayMode == "self" and origin or (pos - dir * FL.DMG.LeadStuds)
	pcall(function()
		PlayerEvent:FireServer("damage", {
			bodyParts = { { "Head", 1 } },
			shotCode = { from, dir },
			pos = pos,
			target = (typeof(target) == "Instance" and target:IsA("Player")) and target or nil,
			damageFactor = 1.5,
		})
	end)
	return true
end

FL.TRACK = { Enabled=false, Full360=true, FOV=200, MaxDist=99999, Mode="all",
	CombatOnly=false, ForceHit=true, ForceRate=0.12, ShowCircle=true, ShowLine=true,
	Color=Color3.fromRGB(255,90,90), Target=nil, _acc=0, _conn=nil }

function FL.TRACK.Pass(plr)
	if not plr or plr == LocalPlayer then return false end
	local c = plr.Character
	if not c then return false end
	local h = c:FindFirstChildOfClass("Humanoid")
	if not h or h.Health <= 0 then return false end
	if FL.TRACK.CombatOnly and plr:GetAttribute("CombatMode") ~= true then return false end
	local f = plr.Team and plr.Team.Name or "?"
	if FL.TRACK.Mode == "police" then return f == "Police" end
	if FL.TRACK.Mode == "civilian" then return f == "Civilian" end
	return true
end

local FLTrackGUI, FLTrackCircle, FLTrackStroke, FLTrackLine
local function FLTrackBuild()
	if FLTrackGUI and FLTrackGUI.Parent then return end
	local pg = UI_PARENT or resolveGuiParent()
	if not pg then return end
	FLTrackGUI = Instance.new("ScreenGui")
	FLTrackGUI.Name = "FL_Track"; FLTrackGUI.IgnoreGuiInset = true
	FLTrackGUI.ResetOnSpawn = false; FLTrackGUI.DisplayOrder = 2
	FLTrackGUI.Parent = pg
	FLTrackCircle = Instance.new("Frame")
	FLTrackCircle.AnchorPoint = Vector2.new(0.5,0.5)
	FLTrackCircle.Position = UDim2.fromScale(0.5,0.5)
	FLTrackCircle.BackgroundTransparency = 1
	FLTrackCircle.Visible = false
	FLTrackCircle.Parent = FLTrackGUI
	Instance.new("UICorner", FLTrackCircle).CornerRadius = UDim.new(1,0)
	FLTrackStroke = Instance.new("UIStroke", FLTrackCircle)
	FLTrackStroke.Thickness = 1; FLTrackStroke.Transparency = 0.35
	FLTrackLine = Instance.new("Frame")
	FLTrackLine.AnchorPoint = Vector2.new(0,0.5)
	FLTrackLine.BorderSizePixel = 0; FLTrackLine.Visible = false
	FLTrackLine.ZIndex = 3; FLTrackLine.Parent = FLTrackGUI
end

local function FLTrackFovPx()
	local cam = Workspace.CurrentCamera
	local h = cam and cam.ViewportSize.Y or 1080
	return FL.TRACK.FOV * (h/1080)
end

local function FLTrackFind()
	local cam = Workspace.CurrentCamera
	if not cam then return nil end
	local vp = cam.ViewportSize
	local center = Vector2.new(vp.X/2, vp.Y/2)
	local camPos = cam.CFrame.Position
	local r = FLTrackFovPx()
	local best, bestD = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if FL.TRACK.Pass(plr) then
			local c = plr.Character
			local p = c and (c:FindFirstChild("Head") or c:FindFirstChild("UpperTorso") or c:FindFirstChild("HumanoidRootPart"))
			if p then
				local dist = (p.Position - camPos).Magnitude
				if dist <= FL.TRACK.MaxDist then
					local sp = cam:WorldToViewportPoint(p.Position)
					local metric
					if FL.TRACK.Full360 then metric = dist
					elseif sp.Z > 0 then
						local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
						if d <= r then metric = d end
					end
					if metric and metric < bestD then
						bestD = metric
						best = { plr = plr, part = p, screen = Vector2.new(sp.X, sp.Y), dist = dist }
					end
				end
			end
		end
	end
	return best
end

local function FLTrackStep(dt)
	if not FL.TRACK.Enabled then
		FL.TRACK.Target = nil
		if FLTrackCircle then FLTrackCircle.Visible = false end
		if FLTrackLine then FLTrackLine.Visible = false end
		return
	end
	FLTrackBuild()
	local t = FLTrackFind()
	FL.TRACK.Target = t
	local cam = Workspace.CurrentCamera
	if not cam then return end
	local vp = cam.ViewportSize
	local center = Vector2.new(vp.X/2, vp.Y/2)
	if FLTrackCircle then
		FLTrackCircle.Visible = FL.TRACK.ShowCircle and not FL.TRACK.Full360 and t ~= nil
		if FLTrackCircle.Visible then
			local rr = FLTrackFovPx()
			FLTrackCircle.Size = UDim2.fromOffset(rr*2, rr*2)
			FLTrackStroke.Color = FL.TRACK.Color
		end
	end
	if FLTrackLine then
		FLTrackLine.Visible = false
		if FL.TRACK.ShowLine and t then
			local d = t.screen - center
			if d.Magnitude >= 1 then
				FLTrackLine.Position = UDim2.fromOffset(center.X, center.Y)
				FLTrackLine.Size = UDim2.fromOffset(d.Magnitude, 1)
				FLTrackLine.Rotation = math.deg(math.atan2(d.Y, d.X))
				FLTrackLine.BackgroundColor3 = FL.TRACK.Color
				FLTrackLine.Visible = true
			end
		end
	end
	if t and FL.TRACK.ForceHit then
		FL.TRACK._acc = FL.TRACK._acc + dt
		local rate = math.max(0.05, tonumber(FL.TRACK.ForceRate) or 0.12)
		if FL.TRACK._acc >= rate then
			FL.TRACK._acc = 0
			pcall(FL.Hit, t.plr)
		end
	end
end

function FL.TRACK.Set(on)
	FL.TRACK.Enabled = on and true or false
	if FL.TRACK.Enabled then
		FLTrackBuild()
		if not FL.TRACK._conn then
			FL.TRACK._conn = RunService.RenderStepped:Connect(function(dt) pcall(FLTrackStep, dt) end)
		end
	else
		if FL.TRACK._conn then FL.TRACK._conn:Disconnect(); FL.TRACK._conn = nil end
		if FLTrackCircle then FLTrackCircle.Visible = false end
		if FLTrackLine then FLTrackLine.Visible = false end
	end
end

FL.TRACER = { Enabled=false, Hold=3, Drift=1.2, Scale=0.5, Count=0 }

local function FLTracerFolder()
	local f = Workspace:FindFirstChild("FL_Tracers")
	if not f then f = Instance.new("Folder"); f.Name = "FL_Tracers"; f.Parent = Workspace end
	return f

local function FLTracerMuzzle()
	local char = LocalPlayer.Character
	if char then
		for _, d in ipairs(char:GetDescendants()) do
			if d:IsA("ModuleScript") and d.Name == "Config" and d.Parent then
				for _, n in ipairs({"Barrel","Muzzle","MuzzlePart","Tip"}) do
					local m = d.Parent:FindFirstChild(n, true)
					if m then
						if m:IsA("BasePart") then return m.Position end
						if m:IsA("Attachment") then return m.WorldPosition end
					end
				end
			end
		end
	end
	local cam = Workspace.CurrentCamera
	return cam and cam.CFrame.Position or nil
end

local function FLSpawnTracer(from, to)
	if typeof(from) ~= "Vector3" or typeof(to) ~= "Vector3" then return end
	if (to-from).Magnitude < 2 then return end
	local f = FLTracerFolder()
	local holder = Instance.new("Folder"); holder.Name = "Tracer"; holder.Parent = f
	FL.TRACER.Count = FL.TRACER.Count + 1
	local function pt(pos)
		local p = Instance.new("Part")
		p.Anchored = true; p.CanCollide = false; p.CanQuery = false
		p.Transparency = 1; p.Size = Vector3.new(0.05,0.05,0.05)
		p.CFrame = CFrame.new(pos); p.Parent = holder
		local a = Instance.new("Attachment", p)
		return p, a
	end
	local p0, a0 = pt(from)
	local p1, a1 = pt(to)
	local function beam(w0, w1, trans, emis)
		local b = Instance.new("Beam")
		b.Attachment0 = a0; b.Attachment1 = a1
		b.Width0 = w0 * FL.TRACER.Scale; b.Width1 = w1 * FL.TRACER.Scale
		b.Segments = 1; b.FaceCamera = true
		b.LightEmission = emis; b.LightInfluence = 0
		b.Color = ColorSequence.new(Color3.new(1,1,1))
		b.Transparency = NumberSequence.new(trans)
		b.Parent = holder
		return b
	end
	local core = beam(0.20, 0.06, 0, 1)
	local mid  = beam(0.55, 0.20, 0.72, 0.8)
	local halo = beam(1.30, 0.45, 0.88, 0.45)
	task.spawn(function()
		task.wait(math.max(0.1, FL.TRACER.Hold))
		if not holder.Parent then return end
		local t = 0
		while t < FL.TRACER.Drift do
			local dt = RunService.Heartbeat:Wait()
			if not holder.Parent then return end
			t = t + dt
			local e = math.clamp(t/FL.TRACER.Drift, 0, 1); e = e*e
			core.Transparency = NumberSequence.new(e)
			mid.Transparency  = NumberSequence.new(0.72 + 0.28*e)
			halo.Transparency = NumberSequence.new(0.88 + 0.12*e)
			local up = Vector3.new(0, 1.4*e, 0)
			p0.CFrame = CFrame.new(from + up)
			p1.CFrame = CFrame.new(to + up*1.8)
		end
		holder:Destroy()
	end)
end

function FL.TRACER.OnBullet(payload)
	if not FL.TRACER.Enabled then return end
	if type(payload) ~= "table" or typeof(payload.pos) ~= "Vector3" then return end
	local from = FLTracerMuzzle()
	if from then pcall(FLSpawnTracer, from, payload.pos) end
end

function FL.TRACER.Set(on)
	FL.TRACER.Enabled = on and true or false
	if FL.TRACER.Enabled then
		FLTracerFolder()
		if not FL.TRACER._hook and hookmetamethod and newcclosure then
			local ok = pcall(function()
				local old = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
					if FL.TRACER.Enabled and getnamecallmethod() == "FireServer"
						and self == PlayerEvent and select(1, ...) == "bullet" then
						pcall(FL.TRACER.OnBullet, select(2, ...))
					end
					return old(self, ...)
				end))
			end)
			FL.TRACER._hook = ok
		end
	else
		local f = Workspace:FindFirstChild("FL_Tracers")
		if f then for _, c in ipairs(f:GetChildren()) do pcall(function() c:Destroy() end) end end
	end
end

				if not r then return end
				if FL.SPIN.Random then FL.SPIN._angle = math.random() * math.pi * 2
				else FL.SPIN._angle = (FL.SPIN._angle + math.rad(FL.SPIN.Speed) * dt) % (math.pi * 2) end
				pcall(function() r.CFrame = CFrame.new(r.Position) * CFrame.Angles(0, FL.SPIN._angle, 0) end)
			end)
		end
	else
		if FL.SPIN._conn then FL.SPIN._conn:Disconnect(); FL.SPIN._conn = nil end
	end
end

FL.TIRE = { Enabled=false, Range=400, Interval=0.15, Count=0, Queue={}, Conn=nil }

local function FLTireTeam(veh)
	local cfg = veh:FindFirstChild("Config")
	local t = cfg and cfg:FindFirstChild("Type")
	local v = t and t.Value
	if v == "Special" then v = "Civilian" end
	return v
end
local function FLTireDriver(veh)
	local cfg = veh:FindFirstChild("Config")
	local d = cfg and cfg:FindFirstChild("LastDrove")
	local p = d and d.Value
	return (typeof(p) == "Instance" and p:IsA("Player")) and p or nil
end
local function FLTireIsTarget(veh)
	if FLTireTeam(veh) == "Police" then return true end
	local d = FLTireDriver(veh)
	if d then
		if d.Team and d.Team.Name == "Police" then return true end
		if d:GetAttribute("CombatMode") == true then return true end
	end
	return false
end
local function FLTireCollect(veh)
	local out = {}
	local th = veh:FindFirstChild("Thrusters")
	if not th then return out end
	for _, t in ipairs(th:GetChildren()) do
		local w = t:FindFirstChild("Wheel")
		local ti = w and w:FindFirstChild("Tire")
		local col = ti and ti:FindFirstChild("WheelCollision")
		if col and col:IsA("BasePart") then
			if not col:GetAttribute("DontPuncture") then
				local dur = col:GetAttribute("Durability")
				if dur == nil or dur ~= 0 then table.insert(out, col) end
			end
		end
	end
	return out
end
function FL.TIRE.Puncture(col)
	if not PlayerEvent or not col or not col.Parent then return false end
	local pos = col.Position
	local cam = Workspace.CurrentCamera
	local eye = cam and cam.CFrame.Position
	local dir = eye and (pos - eye).Unit or Vector3.new(0,0,-1)
	pcall(function()
		PlayerEvent:FireServer("damage", { shotCode = { pos - dir*1.5, dir }, pos = pos, targetTire = col })
	end)
	FL.TIRE.Count = FL.TIRE.Count + 1
	return true
end
function FL.TIRE.Sweep()
	local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not myRoot then return 0 end
	local g = Workspace:FindFirstChild("Gameplay")
	local v = g and g:FindFirstChild("Vehicles")
	if not v then return 0 end
	local q = {}
	for _, veh in ipairs(v:GetChildren()) do
		if veh:IsA("Model") and FLTireIsTarget(veh) then
			local root = veh.PrimaryPart or veh:FindFirstChildWhichIsA("BasePart")
			local dist = root and (root.Position - myRoot.Position).Magnitude or 0
			if not root or dist <= FL.TIRE.Range then
				for _, col in ipairs(FLTireCollect(veh)) do q[#q+1] = col end
			end
		end
	end
	FL.TIRE.Queue = q
	return #q
end
function FL.TIRE.Set(on)
	FL.TIRE.Enabled = on and true or false
	if FL.TIRE.Conn then FL.TIRE.Conn:Disconnect(); FL.TIRE.Conn = nil end
	if not FL.TIRE.Enabled then FL.TIRE.Queue = {} return end
	task.spawn(function()
		while FL.TIRE.Enabled do pcall(FL.TIRE.Sweep); task.wait(FL.TIRE.Interval) end
	end)
	local acc = 0
	FL.TIRE.Conn = RunService.Heartbeat:Connect(function(dt)
		if not FL.TIRE.Enabled then return end
		acc = acc + dt
		if acc < 0.06 then return end
		acc = 0
		local col = table.remove(FL.TIRE.Queue, 1)
		if col and col.Parent then pcall(FL.TIRE.Puncture, col) end
	end)
end

FL.CUFF = { Enabled=false, Ring=30, Margin=8, Cooldown=0.4, OnlyWanted=true,

local function FLCuffSelfRoot()
	local c = LocalPlayer.Character
	return c and (c:FindFirstChild("HumanoidRootPart") or c.PrimaryPart)
end
local function FLCuffNearestPolice()
	local root = FLCuffSelfRoot()
	if not root then return nil, nil end
	local best, bestD = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Team and p.Team.Name == "Police" then
			local ch = p.Character
			local hrp = ch and (ch:FindFirstChild("HumanoidRootPart") or ch.PrimaryPart)
			if hrp then
				local d = (hrp.Position - root.Position).Magnitude
				if d < bestD then bestD = d; best = p end
			end
		end
	end
	return best, bestD
end
function FL.CUFF.PushOut(plr)
	local root = FLCuffSelfRoot()
	if not root then return false end
	local ch = plr and plr.Character
	local hrp = ch and (ch:FindFirstChild("HumanoidRootPart") or ch.PrimaryPart)
	if not hrp then return false end
	local delta = root.Position - hrp.Position
	local away = Vector3.new(delta.X, 0, delta.Z)
	if away.Magnitude < 0.01 then away = Vector3.new(1,0,0) end
	local target = hrp.Position + away.Unit * (FL.CUFF.Ring + FL.CUFF.Margin)
	local ok = false
	if FL_AC and FL_AC.SafePivot then ok = FL_AC.SafePivot(target)
	else ok = pcall(function() root.CFrame = CFrame.new(target) end) end
	if ok then FL.CUFF.Pushes = FL.CUFF.Pushes + 1 end
	return ok
end
function FL.CUFF.Set(on)
	FL.CUFF.Enabled = on and true or false
	if FL.CUFF.Conn then FL.CUFF.Conn:Disconnect(); FL.CUFF.Conn = nil end
	if not FL.CUFF.Enabled then return end
	FL.CUFF.Conn = RunService.Heartbeat:Connect(function()
		if not FL.CUFF.Enabled then return end
		if FL.CUFF.OnlyWanted then
			local w = LocalPlayer:GetAttribute("WantedLevel")
			if not (w and w ~= false) then return end
		end
		local plr, d = FLCuffNearestPolice()
		if not plr or not d or d >= FL.CUFF.Ring then return end
		local now = tick()
		if now - FL.CUFF.LastPush < FL.CUFF.Cooldown then return end
		FL.CUFF.LastPush = now
		FL.CUFF.PushOut(plr)
	end)
end

	local char = LocalPlayer.Character

function FL.WPN.Config()
	local m = FL.WPN.Model()
	if not m then return nil end
	local cfg = m:FindFirstChild("Config")
	if not cfg or not cfg:IsA("ModuleScript") then return nil end
	local ok, t = pcall(require, cfg)
	if ok and type(t) == "table" then return t end
	return nil
end

function FL.WPN.Apply()
	local t = FL.WPN.Config()
	if not t then return false end
	if not FL.WPN._orig[t] then
		FL.WPN._orig[t] = { RPM = rawget(t,"RPM"), SHOOT_MODE = rawget(t,"SHOOT_MODE"), BULLET_DISTANCE = rawget(t,"BULLET_DISTANCE") }
	end
	if FL.WPN.RPMOn then pcall(rawset, t, "RPM", math.clamp(FL.WPN.RPM, 60, FL.WPN.RPMMax)) end
	if FL.WPN.RangeOn then pcall(rawset, t, "BULLET_DISTANCE", math.max(600, FL.WPN.Range)) end
	if FL.WPN.ModeOn then pcall(rawset, t, "SHOOT_MODE", FL.WPN.Mode) end
	return true
end

function FL.WPN.Restore()
	for t, o in pairs(FL.WPN._orig) do
		if type(t) == "table" then
			if o.RPM ~= nil then pcall(rawset, t, "RPM", o.RPM) end
			if o.SHOOT_MODE ~= nil then pcall(rawset, t, "SHOOT_MODE", o.SHOOT_MODE) end
			if o.BULLET_DISTANCE ~= nil then pcall(rawset, t, "BULLET_DISTANCE", o.BULLET_DISTANCE) end
		end
	end
	FL.WPN._orig = {}
end

local function FLWPN_Tick()
	if FL.WPN.RPMOn or FL.WPN.RangeOn or FL.WPN.ModeOn then pcall(FL.WPN.Apply) end
	if FL.WPN.InfiniteAmmo then
		local m = FL.WPN.Model()
		if m then
			local cfg = m:FindFirstChild("Config")
			if cfg then
				for _, n in ipairs({"Ammo","TotalAmmo"}) do
					local a = cfg:FindFirstChild(n)
					if a and a:IsA("ValueBase") then pcall(function() a.Value = FL.WPN.AmmoLock end) end
				end
			end
		end
	end
end

function FL.WPN.Sync()
	local need = FL.WPN.InfiniteAmmo or FL.WPN.RPMOn or FL.WPN.RangeOn or FL.WPN.ModeOn
	if need and not FL.WPN._conn then
		FL.WPN._conn = RunService.Heartbeat:Connect(function() pcall(FLWPN_Tick) end)
		pcall(FL.WPN.Apply)
	elseif not need and FL.WPN._conn then
		FL.WPN._conn:Disconnect(); FL.WPN._conn = nil
		pcall(FL.WPN.Restore)
	end
end

local function espAllowed(player)
	if not ESP.enabled then return false end

end

local function ESP_BBox(body)
	local cam = Workspace.CurrentCamera
	if not cam then return nil end
	local minX, minY, maxX, maxY = math.huge, math.huge, -math.huge, -math.huge
	local any = false
	for _, p in pairs(body) do
		local sz = p.Size / 2
		for sx = -1, 1, 2 do for sy = -1, 1, 2 do for sz2 = -1, 1, 2 do
			local c = p.Position + Vector3.new(sz.X*sx, sz.Y*sy, sz.Z*sz2)
			local v, on = cam:WorldToViewportPoint(c)
			if on then
				any = true
				if v.X < minX then minX = v.X end
				if v.Y < minY then minY = v.Y end
				if v.X > maxX then maxX = v.X end
				if v.Y > maxY then maxY = v.Y end
			end
		end end end
	end
	if not any then return nil end
	return Vector2.new(minX, minY), Vector2.new(maxX-minX, maxY-minY)
end

function removeESP(player)
	local t = ESP.trackers[player]
	if not t then return end
	for _, v in pairs(t.R) do
		if typeof(v) == "Instance" then pcall(function() v:Destroy() end)
		elseif type(v) == "table" then
			for _, l in pairs(v) do if typeof(l)=="Instance" then pcall(function() l:Destroy() end) end end
		end
	end
	if t.conn then pcall(function() t.conn:Disconnect() end) end
	ESP.trackers[player] = nil
end

function createESP(player)
	removeESP(player)
	if not espAllowed(player) then return end
	local ui = UI_PARENT or resolveGuiParent()
	if not ui then return end
	local R = ESP_MakeRender()
	R.Frame.Parent = ui
	R.OOV.Parent = ui
	for i=1,#R.Skeleton do R.Skeleton[i].Parent = ui end

	local conn = RunService.Heartbeat:Connect(function()
		local char = player.Character
		if not char or not espAllowed(player) then
			R.Frame.Visible = false; R.OOV.Visible = false
			for i=1,#R.Skeleton do R.Skeleton[i].Visible = false end
			return
		end
		local hum = char:FindFirstChildOfClass("Humanoid")
		local root = char:FindFirstChild("HumanoidRootPart")
		if not hum or not root then return end
		local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
		local dist = myRoot and (myRoot.Position - root.Position).Magnitude or 0
		if dist > ESP.maxDist then
			R.Frame.Visible = false; R.OOV.Visible = false
			for i=1,#R.Skeleton do R.Skeleton[i].Visible = false end
			return
		end
		local body = ESP_Body(char)
		local bbMin, bbSz = ESP_BBox(body)
		if not bbMin then
			R.Frame.Visible = false; R.OOV.Visible = false
			for i=1,#R.Skeleton do R.Skeleton[i].Visible = false end
			return
		end
		local fug = isFugitive(player)
		local team = player.Team and player.Team.Name
		local baseColor = fug and Color3.fromRGB(255,60,60) or TeamColors[team] or ESP.boxColor

		R.Frame.Visible = true

		R.Box.Size = UDim2.new(1,2,1,2)
		R.Box.Visible = ESP.box
		R.BoxOutline.Color = baseColor
		R.BoxOutline.Thickness = ESP.boxThick
		R.BoxFill.Visible = ESP.box and ESP.fill
		R.BoxFill.BackgroundColor3 = ESP.fillColor
		R.BoxFill.BackgroundTransparency = ESP.fillTrans
		R.BoxFillGrad.Color = ColorSequence.new(ESP.fillColor, ESP.fillColor)

		local hp = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
		R.HPBack.Visible = ESP.health
		R.HPBar.Visible = ESP.health
		R.HPText.Visible = ESP.health and hp < 1
		if ESP.health then
			R.HPBack.Position = UDim2.new(0,-6,0,0)
			R.HPBack.Size = UDim2.new(0,2,1,0)
			R.HPBar.Position = UDim2.new(0,0,1-hp,0)
			R.HPBar.Size = UDim2.new(1,0,hp,0)
			if ESP.healthType == "Gradient" then
				R.HPBar.BackgroundColor3 = Color3.new(1,1,1)
				R.HPGrad.Enabled = true
				R.HPGrad.Color = ColorSequence.new(Color3.fromRGB(255,80,80), Color3.fromRGB(80,255,80))
			else
				R.HPGrad.Enabled = false
				R.HPBar.BackgroundColor3 = ESP_LerpColor(Color3.fromRGB(255,60,60), Color3.fromRGB(60,255,60), hp)
			end
			R.HPText.Text = tostring(math.ceil(hum.Health))
			R.HPText.Position = UDim2.new(0,-22,1,0)
		end

		if ESP.name then
			local nm = player.DisplayName ~= "" and player.DisplayName or player.Name
			local stars = tonumber(player:GetAttribute("WantedLevel")) or 0
			if stars > 0 then nm = nm.." "..string.rep("*", math.min(stars, 6)) end

			R.NameLabel.Position = UDim2.new(0.5,0,0,-16)
			R.NameLabel.TextColor3 = baseColor
			R.NameLabel.Visible = true
		else R.NameLabel.Visible = false end

		if ESP.distance then
			R.Distance.Text = string.format("[%dM]", math.floor(dist))
			R.Distance.Position = UDim2.new(0.5,0,1,2)
			R.Distance.TextColor3 = baseColor
			R.Distance.Visible = true
		else R.Distance.Visible = false end

		if ESP.weapon then
			local tool = char:FindFirstChildOfClass("Tool")
			if tool then
				R.Weapon.Text = tool.Name
				R.Weapon.Position = UDim2.new(0.5,0,1,14)
				R.Weapon.TextColor3 = baseColor
				R.Weapon.Visible = true
			else R.Weapon.Visible = false end
		else R.Weapon.Visible = false end

		if ESP.skeleton then
			local cam = Workspace.CurrentCamera
			for i, pair in ipairs(ESP_SkeletonJoints) do
				local line = R.Skeleton[i]
				local a, b = body[pair[1]], body[pair[2]]
				if a and b and cam then
					local pa, oa = cam:WorldToViewportPoint(a.Position)
					local pb, ob = cam:WorldToViewportPoint(b.Position)
					if oa and ob then

						else line.Visible = false end
					else line.Visible = false end

			hl.Enabled = true

			local center = Vector2.new(vp.X/2, vp.Y/2)
			local rel = cam.CFrame:PointToObjectSpace(root.Position)
			local ang = math.atan2(-rel.Y, rel.X)
			local dir = Vector2.new(math.cos(ang), math.sin(ang))
			local radius = ESP.oovDist / 400
			local pa = (dir * vp.X * radius) + center
			local sz = ESP.oovSize
			local function rot(v, r) return Vector2.new(v.X*math.cos(r)-v.Y*math.sin(r), v.X*math.sin(r)+v.Y*math.cos(r)) end
			local p2 = pa - rot(dir, 0.45) * sz
			local p3 = pa - rot(dir, -0.45) * sz
			local function setLine(line, from, to)
				local d = to - from
				if d.Magnitude < 0.5 then line.Visible = false return end
				line.Position = UDim2.fromOffset(from.X, from.Y)
				line.Size = UDim2.fromOffset(d.Magnitude, 2)
				line.Rotation = math.deg(math.atan2(d.Y, d.X))
				line.BackgroundColor3 = ESP.oovColor
				line.Visible = true
			end
			setLine(R.OOV_L, pa, p2)
			setLine(R.OOV_R, pa, p3)
			if ESP.oovStyle == "Concave" then
				local pd = pa - dir * (sz/1.6)
				setLine(R.OOV_M, p2, pd)
			else R.OOV_M.Visible = false end
		elseif ESP.tracer and on then
			R.OOV.Visible = true
			local vp = cam.ViewportSize
			local origin
			if ESP.tracerOrigin == "屏幕中心" then origin = Vector2.new(vp.X/2, vp.Y/2)
			elseif ESP.tracerOrigin == "屏幕顶部" then origin = Vector2.new(vp.X/2, 0)
			else origin = Vector2.new(vp.X/2, vp.Y) end
			local to = Vector2.new(sp, select(2, cam:WorldToViewportPoint(root.Position)))
			local dd = to - origin
			if dd.Magnitude >= 1 then
				R.OOV_L.Position = UDim2.fromOffset(origin.X, origin.Y)
				R.OOV_L.Size = UDim2.fromOffset(dd.Magnitude, 1)
				R.OOV_L.Rotation = math.deg(math.atan2(dd.Y, dd.X))
				R.OOV_L.BackgroundColor3 = baseColor
				R.OOV_L.Visible = true
				R.OOV_R.Visible = false
				R.OOV_M.Visible = false
			end
		else
			R.OOV.Visible = false
		end
	end)
	ESP.trackers[player] = { R = R, conn = conn }
end

function refreshESP()
	for p in pairs(ESP.trackers) do
		if not espAllowed(p) then removeESP(p) end
	end
	if ESP.enabled then
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LocalPlayer and espAllowed(p) and not ESP.trackers[p] then createESP(p) end
		end
	end
end

end)

local PoliceConfig = { range = 200, delay = 0.5, combatCheck = false, teleport = false }
local autoCuffThread
local function startAutoCuff()
	if autoCuffThread then return end
	autoCuffThread = task.spawn(function()
		while State.autoCuff do
			local _, _, myRoot = GetCharacter(LocalPlayer)
			if myRoot and PlayerFunc then
				for _, player in ipairs(Players:GetPlayers()) do
					if player ~= LocalPlayer and IsAlive(player) and inCombat(player, PoliceConfig.combatCheck) then
						local _, _, root = GetCharacter(player)
						if root and (root.Position - myRoot.Position).Magnitude <= PoliceConfig.range then
							pcall(function() PlayerFunc:InvokeServer("handcuff", player, false) end)
						end
					end
				end
			end
			task.wait(PoliceConfig.delay + math.random() * 0.2)
		end
		autoCuffThread = nil
	end)
end

local function getPromptPosition(prompt)
	if not prompt or not prompt.Parent then return end
	local parent = prompt.Parent
	if parent:IsA("BasePart") then return parent.Position end
	if parent:IsA("Attachment") then return parent.WorldPosition end
	if parent:IsA("Model") then
		local part = parent.PrimaryPart or parent:FindFirstChildWhichIsA("BasePart")
		if part then return part.Position end
	end
end
local function firePrompt(prompt)
	if not prompt then return end
	pcall(function()
		if fireproximityprompt then fireproximityprompt(prompt, 0)
		else prompt.HoldDuration = 0; prompt:InputHoldBegin(); task.wait(0.1); prompt:InputHoldEnd() end
	end)
end
task.spawn(function()
	while true do
		if State.autoMoney then
			local _, _, root = GetCharacter(LocalPlayer)
			if root then
				local bestPrompt, bestDist
				for _, obj in ipairs(Workspace:GetDescendants()) do
					if obj:IsA("ProximityPrompt") and (obj.Name == "CashDrop" or obj.Name == "GetItem") then
						local pos = getPromptPosition(obj)
						if pos then
							local dist = (root.Position - pos).Magnitude
							if not bestDist or dist < bestDist then bestDist = dist; bestPrompt = obj end
						end
					end
				end
				if bestPrompt and bestDist and bestDist < 60 then
					if bestDist > 8 then safeTeleport(getPromptPosition(bestPrompt)); task.wait(0.3) end
					firePrompt(bestPrompt)
				end
			end
		end
		task.wait(0.3 + math.random() * 0.2)
	end
end)
